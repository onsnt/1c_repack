// Укажите путь к каталогу на диске, где лежат обработанные файлы .epf / .erf
ПутьККаталогу = "C:\Temp\ExtProcessed\";

Если Не НайтиФайлы(ПутьККаталогу, "").Количество() > 0 Тогда
    ВызватьИсключение "Каталог с файлами не найден или пуст: " + ПутьККаталогу;
КонецЕсли;

ИмяСправочника = "ДополнительныеОтчетыИОбработки";

// Ищем все файлы обработок и отчетов
МассивФайлов = НайтиФайлы(ПутьККаталогу, "*.epf");
Для Каждого ФайлЭлемент Из НайтиФайлы(ПутьККаталогу, "*.erf") Цикл

СчетчикОбновленных = 0;

Сообщить("[*] Найдено файлов для загрузки в базу: " + МассивФайлов.Количество());

Для Каждого ФайлСтрока Из МассивФайлов Цикл
    
    ПолныйПуть = ФайлСтрока.ПолноеИмя;
    ИмяБезРасширения = ФайлСтрока.ИмяБезРасширения;
    
    // Ожидаемый формат имени (если выгрузка шла нашим скриптом): Название_GUID.epf
    // Попробуем извлечь часть GUID из хвоста имени файла (последние 6-36 символов) или искать по наименованию
    ПоискПоНаименованию = ИмяБезРасширения;
    
    // Ищем существующий элемент в справочнике
    // В БСП основным реквизитом для поиска является Наименование или ИмяОбъекта
    НайденнаяСсылка = Справочники[ИмяСправочника].НайтиПоНаименованию(ПоискПоНаименованию);
    
    НачалоТранзакции();
    Попытка
        
        Если Не ЗначениеЗаполнено(НайденнаяСсылка) Тогда
            // Элемент не найден — создаем новый
            ОбъектЭлемент = Справочники[ИмяСправочника].СоздатьЭлемент();
            ОбъектЭлемент.Наименование = ПоискПоНаименованию;
            
            // Заполняем базовые служебные поля БСП (если требуются конфигурацией)
            Если Метаданные.Справочники[ИмяСправочника].Реквизиты.Найти("ПутьКФайлу") <> Неопределено Тогда
                ОбъектЭлемент.ПутьКФайлу = ФайлСтрока.Имя;
            КонецЕсли;
            
            РежимСоздания = Истина;
        Иначе
            // Элемент найден — обновляем существующий
            ОбъектЭлемент = НайденнаяСсылка.ПолучитьОбъект();
            РежимСоздания = Ложь;
        КонецЕсли;
        
        // Загружаем двоичные данные файла в хранилище
        ДвоичныеДанныеФайла = Новый ДвоичныеДанные(ПолныйПуть);
        
        Если Метаданные.Справочники[ИмяСправочника].Реквизиты.Найти("ХранилищеХранилища") <> Неопределено Тогда
            ОбъектЭлемент.ХранилищеХранилища = Новый ХранилищеЗначения(ДвоичныеДанныеФайла, Новый СжатиеДанных(9));
        ИначеЕсли Метаданные.Справочники[ИмяСправочника].Реквизиты.Найти("ХранилищеОбработки") <> Неопределено Тогда
            ОбъектЭлемент.ХранилищеОбработки = Новый ХранилищеЗначения(ДвоичныеДанныеФайла, Новый СжатиеДанных(9));
        КонецЕсли;
        
        // Установка вида (Отчет / Обработка)
        Если Метаданные.Справочники[ИмяСправочника].Реквизиты.Найти("Вид") <> Неопределено Тогда
            ОбъектЭлемент.Вид = ?(Вкр(ФайлСтрока.Расширение) = ".ERF", Перечисления.ВидыДополнительныхОтчетовИОбработок.Отчет, Перечисления.ВидыДополнительныхОтчетовИОбработок.Обработка);
        КонецЕсли;
        
        ОбъектЭлемент.Записать();
        ФиксацияТранзакции();
        
        Если РежимСоздания Тогда
            СчетчикСозданных = СчетчикСозданных + 1;
            Сообщить("  [+] Создан новый элемент: " + ПоискПоНаименованию);
        Иначе
            СчетчикОбновленных = СчетчикОбновленных + 1;
            Сообщить("  [~] Обновлен существующий элемент: " + ПоискПоНаименованию);
        КонецЕсли;
        
    Исключение
        ОтменитьТранзакцию();
        СчетчикОшибок = СчетчикОшибок + 1;
        Сообщить("  [!] Ошибка обработки файла '" + ФайлСтрока.Имя + "': " + ОписаниеОшибки(), СтатусСообщения.ОченьВажное);
    ПопыткаКонец;
    
КонецЦикла;

Сообщить("=" * 50);
