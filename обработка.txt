СчетчикВыгруженных = 0;

    ИмяСправочника = "ДополнительныеОтчетыИОбработки";

    ИмяСправочника = "ВнешниеОбработки";

Если ПустаяСтрока(ИмяСправочника) Тогда
    ВызватьИсключение "В конфигурации не найдены справочники внешних обработок!";

Сообщить("[*] Сканирование справочника: " + ИмяСправочника);

    // Проверяем наличие двоичных данных (в БСП это может быть ХранилищеХранилища или ХранилищеОбработки)

        ДвоичныеДанныеФайла = Выборка.ХранилищеОбработки.Получить();
    КонецЕсли;
    
    Если ТипЗнч(ДвоичныеДанныеФайла) = Тип("ДвоичныеДанные") Тогда
        
        // Определяем расширение (.epf или .erf) по признаку отчета или обработки, если есть реквизит
        Расширение = ".epf";
        Если Метаданные.Справочники[ИмяСправочника].Реквизиты.Найти("Вид") <> Неопределено Тогда
            Если Строка(Выборка.Вид) = "Отчет" Тогда
                Расширение = ".erf";
            КонецЕсли;
        КонецЕсли;
        
        // Формируем безопасное имя файла
        ИмяФайла = ОбщегоНазначенияКлиентСервер.ОчиститьИмяФайла(Выборка.Наименование);

        КонецЕсли;
        
        ПолныйПутьФайла = ПутьДляВыгрузки + ИмяФайла + "_" + Сред(СокрЛП(Выборка.УникальныйИдентификатор()), 1, 6) + Расширение;
