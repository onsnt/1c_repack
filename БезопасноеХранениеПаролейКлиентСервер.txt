РегистрыСведений -> ПараметрыПодключенияВнешнихСистем.

На вкладке Основные:

    Режим записи: Независимая.

На вкладке Данные:

    Измерения:

        ИдентификаторСистемы (Тип: Строка, Длина: 50) — ключ внешней системы (например, "SiteAPI", "TelegramBot", "WMS").

    Ресурсы:

        Логин (Тип: Строка, Длина: 100) — при необходимости можно тоже шифровать, но обычно критичен только пароль.

        ПарольЗашифрованный (Тип: ХранилищеЗначения) — здесь будет храниться зашифрованный бинарный массив (или строка).

        ДополнительныеПараметры (Тип: ХранилищеЗначения) — для JSON/структуры прочих настроек (адреса, порты).

Функция ПолучитьПараметрыСистемы(ИмяСистемы) Экспорт
    
    Результат = Новый Структура("Логин, Пароль, Адрес", "", "", "");
    
    // Читаем из регистра сведений
    Набор = РегистрыСведений.ПараметрыПодключенияВнешнихСистем.СоздатьНаборЗаписей();
    Набор.Отбор.ИдентификаторСистемы.Установить(ИмяСистемы);
    Набор.Прочитать();
    
    Если Набор.Количество() > 0 Тогда
        Запись = Набор[0];
        Результат.Логин = Запись.Логин;
        
        // Расшифровываем пароль
        ЗашифрованныеДанные = Запись.ПарольЗашифрованный.Получить();
        Если ТипЗнч(ЗашифрованныеДанные) = Тип("ДвоичныеДанные") Тогда
            Результат.Пароль = РасшифроватьДанные(ЗашифрованныеДанные);
        КонецЕсли;
        
        // Доп параметры (если хранятся в структуре)
        ДопПараметры = Запись.ДополнительныеПараметры.Получить();
        Если ТипЗнч(ДопПараметры) = Тип("Структура") Тогда
            Результат.Вставить("Дополнительно", ДопПараметры);
        КонецЕсли;
    КонецЕсли;
    
    Возврат Результат;
    
КонецФункции

Процедура ЗаписатьПараметрыСистемы(ИмяСистемы, Логин, ОткрытыйПароль, ДопПараметры = Неопределено) Экспорт
    
    МенеджерЗаписи = РегистрыСведений.ПараметрыПодключенияВнешнихСистем.СоздатьМенеджерЗаписи();
    МенеджерЗаписи.ИдентификаторСистемы = ИмяСистемы;
    МенеджерЗаписи.Прочитать();
    
    МенеджерЗаписи.Логин = Логин;
    
    // Шифруем пароль перед записью
    ЗашифрованныеДвоичныеДанные = ЗашифроватьДанные(ОткрытыйПароль);
    МенеджерЗаписи.ПарольЗашифрованный = Новый ХранилищеЗначения(ЗашифрованныеДвоичныеДанные);
    
    Если ДопПараметры <> Неопределено Тогда
        МенеджерЗаписи.ДополнительныеПараметры = Новый ХранилищеЗначения(ДопПараметры);
    КонецЕсли;
    
    МенеджерЗаписи.Записать();
    
КонецПроцедуры
